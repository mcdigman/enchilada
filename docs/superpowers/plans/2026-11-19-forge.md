```python
        duration_s=DURATION_S,
```

```python
"""Load a window of a Mojito-layout L1 file and build its noise covariance.

Both forge notebooks import this module, so the data side of a block is
written once. `load_window` goes through enchilada's loader and decimator and
lands on the grid a block will be registered on; `analytic_covariance` turns
an analytic one-sided PSD into a `Covariance` on that grid. Nothing here is
specific to the placeholder block.
"""

import os
from collections.abc import Callable, Mapping

import numpy as np

from enchilada import Covariance, L1Data, decimate, load_mojito_l1, transform

PSD = Callable[[np.ndarray], np.ndarray]
"""A one-sided PSD: frequencies in Hz (an array) -> PSD values (an array)."""

PROBE_DURATION_S = 60.0
"""Length of the one-channel probe load that reads the file's sampling interval."""


def load_window(
    path: str | os.PathLike[str],
    *,
    factor: int,
    channels: tuple[str, ...] = ("A", "E", "T"),
    start_offset_s: float = 0.0,
    duration_s: float | None = None,
    domain: str = "time",
    num_frequency_divisions: int | None = None,
    num_time_divisions: int | None = None,
) -> L1Data:
    """Load `channels` from a Mojito-layout file, decimate by `factor` and
    return the window in `domain`.

    Time and frequency windows take `duration_s` (native seconds from
    `start_offset_s`). A WDM window takes both division counts instead and
    derives its native duration, `factor * Nf * Nt * dt`, so the decimated
    record has exactly `Nf * Nt` samples; `num_time_divisions` must be even.
    The first and last `enchilada.resample.edge_samples(factor)` samples (40)
    of the decimated record carry the anti-alias filter's edge transient;
    nothing is trimmed.
    """
    channels = tuple(channels)
    if domain not in ("time", "frequency", "wdm"):
        raise ValueError(f"domain must be 'time', 'frequency' or 'wdm', got {domain!r}")
    if domain == "wdm":
        if num_frequency_divisions is None or num_time_divisions is None:
            raise ValueError(
                "a WDM window needs num_frequency_divisions and num_time_divisions"
            )
        if duration_s is not None:
            raise ValueError(
                "a WDM window derives its duration from the division counts; "
                "leave duration_s as None"
            )
        if num_time_divisions % 2:
            raise ValueError(
                f"num_time_divisions must be even, got {num_time_divisions}"
            )
        probe = load_mojito_l1(
            path, channels=channels[:1], duration_s=PROBE_DURATION_S, orbit=False
        )
        count = factor * num_frequency_divisions * num_time_divisions
        # The loader floors duration_s / dt; the extra half sample keeps roundoff
        # from dropping the last sample.
        duration_s = (count + 0.5) * probe.dt
    elif duration_s is None:
        raise ValueError(f"a {domain} window needs duration_s")
    elif num_frequency_divisions is not None or num_time_divisions is not None:
        raise ValueError("division counts apply only to domain='wdm'")
    native = load_mojito_l1(
        path, channels=channels, start_offset_s=start_offset_s, duration_s=duration_s
    )
    data = decimate(native, factor)
    if domain == "time":
        return data
    if domain == "frequency":
        return transform(data, "frequency")
    return transform(
        data,
        "wdm",
        num_frequency_divisions=num_frequency_divisions,
        num_time_divisions=num_time_divisions,
    )


def white_psd(level: float) -> PSD:
    """One-sided PSD equal to `level` at every frequency.

    White noise of per-sample variance `sigma**2` at sample rate `fs` has
    `level = 2 * sigma**2 / fs`.
    """
    if not np.isfinite(level) or level <= 0:
        raise ValueError(f"level must be finite and positive, got {level!r}")

    def psd(f: np.ndarray) -> np.ndarray:
        return np.full(np.shape(f), level, dtype=float)

    return psd


def analytic_covariance(data: L1Data, psd: PSD | Mapping[str, PSD]) -> Covariance:
    """A `Covariance` on `data`'s grid from a one-sided PSD, channels independent.

    `psd` is one callable for every channel or a mapping from channel name to
    callable. With `X = dt * rfft(x)` a stationary PSD `S` gives rfft-bin
    variance `(Tobs / 2) * S(f_k)`; `S` is evaluated at `max(f_k, df)`, so the
    DC bin takes the first bin's value. A WDM cell at band frequency `f` has
    variance `S(f) / 2`; row 0 holds DC (evaluated at the band spacing) in its
    even columns and the Nyquist band in its odd columns, and the PSD is
    treated as flat across a band. The time grid has one variance per sample,
    `S * fs / 2`, so only a white PSD is accepted there.
    """
    channels = tuple(data.channel_names)
    if callable(psd):
        per_channel = {name: psd for name in channels}
    else:
        missing = [name for name in channels if name not in psd]
        if missing:
            raise ValueError(f"psd mapping has no entry for channels {missing}")
        per_channel = {name: psd[name] for name in channels}
    bins = np.maximum(np.fft.rfftfreq(data.N, data.dt), data.df)
    pairs: dict[str, np.ndarray] = {}
    if data.data_domain == "frequency":
        for name, fn in per_channel.items():
            pairs[name] = 0.5 * data.Tobs * np.asarray(fn(bins), dtype=float)
    elif data.data_domain == "wdm":
        assert data.wdm_grid is not None
        rows, columns = data.wdm_grid.array_shape
        band = np.arange(rows, dtype=float)[:, None] * np.ones((1, columns))
        band[0, 0::2] = 1.0  # DC, evaluated at the band spacing
        band[0, 1::2] = rows  # the Nyquist band
        frequencies = band * data.fs / (2 * rows)
        for name, fn in per_channel.items():
            pairs[name] = 0.5 * np.asarray(fn(frequencies), dtype=float)
    else:
        for name, fn in per_channel.items():
            values = np.asarray(fn(bins), dtype=float)
            if not np.allclose(values, values[0], rtol=1e-9, atol=0.0):
                raise ValueError(
                    f"psd for channel {name!r} is not white: a coloured PSD has "
                    "no per-sample covariance on the time grid; register the "
                    "block on the frequency or WDM grid, or use white_psd"
                )
            pairs[name] = np.full(data.N, 0.5 * values[0] * data.fs)
    return Covariance.like(data, pairs)
```

```python
"""ForgeBlock: the skeleton to fill in.

The methods marked `# EDIT` are the model: `_template` renders the signal on
the block's grid, `initial_sample` draws or renders the start, `sample`
advances the state; `_check_residual` and `_check_covariance` say what the
model needs from its inputs. The placeholder is a sinusoid of one amplitude
at `frequency_hz`, rendered natively on the frequency grid, with a Gaussian
prior on the amplitude, so `sample` is an exact conjugate Gaussian draw: the
skeleton runs green on noise-only data with no tuning and every notebook cell
shows real numbers. Replace the model; keep the shape of the contract.
"""

from dataclasses import dataclass, field

import numpy as np

from enchilada import BlockState, Covariance, L1Data


def rfft_weights(num_time_samples: int) -> np.ndarray:
    """How often each rfft bin counts in a real signal's inner product: once at
    DC and at Nyquist (even `num_time_samples` only), twice elsewhere."""
    weights = np.full(num_time_samples // 2 + 1, 2.0)
    weights[0] = 1.0
    if num_time_samples % 2 == 0:
        weights[-1] = 1.0
    return weights


@dataclass
class ForgeBlock:
    """A sinusoid of one amplitude on the frequency grid; replace the model.

    `prior_std` is the standard deviation of the zero-mean Gaussian prior on
    the amplitude, in the data's units (fractional frequency). The default is
    wide next to the likelihood on a day of Mojito-level data, so the
    posterior is the data's; tighten it when the data barely constrain the
    amplitude. `start_amplitude` renders a chosen start instead of a prior
    draw.
    """

    name: str = "forge"
    frequency_hz: float = 1e-3
    prior_std: float = 1e-17
    start_amplitude: float | None = None

    @dataclass
    class State:
        """Diagnostics owned by the block; never published to the ledger."""

        num_sample_calls: int = 0
        last_posterior_mean: float | None = None
        last_posterior_std: float | None = None

    state: State = field(default_factory=State, init=False)

    def __post_init__(self) -> None:
        if not self.name:
            raise ValueError("name must be a non-empty string")
        if not np.isfinite(self.frequency_hz) or self.frequency_hz <= 0:
            raise ValueError("frequency_hz must be finite and positive")
        if not np.isfinite(self.prior_std) or self.prior_std <= 0:
            raise ValueError("prior_std must be finite and positive")
        if self.start_amplitude is not None and not np.isfinite(self.start_amplitude):
            raise ValueError("start_amplitude must be finite")

    # ---- checks: keep these honest about what the model needs ----------------

    def _check_residual(self, conditional_residual: L1Data) -> None:
        # EDIT: the grid your template is rendered on, and anything else the
        # model assumes about the data.
        if conditional_residual.data_domain != "frequency":
            raise ValueError(
                f"{self.name}: register this block with data_domain='frequency'; "
                f"it received {conditional_residual.data_domain!r} data"
            )
        if self.frequency_hz >= conditional_residual.fny:
            raise ValueError(
                f"{self.name}: frequency_hz = {self.frequency_hz} is not below the "
                f"Nyquist frequency {conditional_residual.fny} Hz"
            )

    def _check_covariance(
        self, conditional_residual: L1Data, covariance: Covariance | None
    ) -> Covariance:
        # EDIT: what your likelihood needs from the noise model.
        if covariance is None:
            raise ValueError(f"{self.name}: the likelihood needs a noise covariance")
        covariance.check_compatible(conditional_residual)
        channels = conditional_residual.channel_names
        if any(
            np.any(covariance.pair(first, second))
            for first in channels
            for second in channels
            if first != second
        ):
            raise ValueError(
                f"{self.name}: the likelihood assumes independent channels"
            )
        return covariance

    # ---- the model -------------------------------------------------------------

    def _basis(self, conditional_residual: L1Data) -> np.ndarray:
        """Unit-amplitude sinusoid at `frequency_hz` as `dt * rfft`, the same in
        every channel."""
        times_s = (
            conditional_residual.start_time_gps
            + np.arange(conditional_residual.num_time_samples) / conditional_residual.fs
        )
        return conditional_residual.dt * np.fft.rfft(
            np.sin(2 * np.pi * self.frequency_hz * times_s)
        )

    def _template(self, conditional_residual: L1Data, amplitude: float) -> BlockState:
        # EDIT: render your waveform on the block's grid.
        self._check_residual(conditional_residual)
        basis = self._basis(conditional_residual)
        return conditional_residual.block_state(
            {name: amplitude * basis for name in conditional_residual.channel_names},
            parameters={"amplitude": float(amplitude)},
        )

    def initial_sample(
        self,
        conditional_residual: L1Data,
        covariance: Covariance | None,
        *,
        rng: np.random.Generator,
    ) -> BlockState:
        # EDIT: your start, drawn from your prior or rendered from configuration.
        amplitude = self.start_amplitude
        if amplitude is None:
            amplitude = rng.normal(0.0, self.prior_std)
        return self._template(conditional_residual, amplitude)

    def sample(
        self,
        conditional_residual: L1Data,
        covariance: Covariance | None,
        current_block_state: BlockState,
        *,
        rng: np.random.Generator,
    ) -> BlockState:
        # EDIT: your sampler. Return a complete state after acceptance and
        # rejection alike. This placeholder ignores current_block_state because
        # it draws the exact conjugate posterior of the amplitude.
        self._check_residual(conditional_residual)
        noise = self._check_covariance(conditional_residual, covariance)
        basis = self._basis(conditional_residual)
        weights = rfft_weights(conditional_residual.num_time_samples)
        precision = 1.0 / self.prior_std**2
        weighted_data = 0.0
        for name in conditional_residual.channel_names:
            scale = weights / noise.pair(name, name)
            precision += float(np.sum(scale * np.abs(basis) ** 2))
            weighted_data += float(
                np.sum(
                    scale
                    * np.real(np.conj(basis) * conditional_residual.channel_data[name])
                )
            )
        mean = weighted_data / precision
        std = precision**-0.5
        self.state.num_sample_calls += 1
        self.state.last_posterior_mean = mean
        self.state.last_posterior_std = std
        return self._template(conditional_residual, rng.normal(mean, std))
```

- [ ] **Step 2: Write `forge/README.md`**

````markdown
# Forge

A workbench for building an enchilada `Block` against the inputs a block
really receives: a conditional residual on a chosen grid and a `Covariance`
on that grid, loaded from a simulated Mojito-layout L1 file. Copy this folder
once per block and fill in the skeleton. The tutorial that explains the
contract is [`examples/create_block.ipynb`](../examples/create_block.ipynb);
this folder assumes you have read it.

## The three steps

1. **Fill in `block.py`.** `ForgeBlock` has three methods marked `# EDIT`:
   `_template` renders the signal on the block's grid, `initial_sample`
   draws or renders the start, `sample` advances the state. Two checks,
   `_check_residual` and `_check_covariance`, say what the model needs from
   its inputs. The placeholder is a sinusoid of one amplitude on the
   frequency grid with a conjugate Gaussian sampler, so everything runs
   before you change a line.
2. **Run `block.ipynb`.** It loads a window of the simulated file, builds the
   covariance, calls `initial_sample` and `sample` by hand with asserts on
   the returned state, and runs `check_block`. Edits to `block.py` take
   effect cell by cell.
3. **Run `wheel.ipynb`.** A `Wheel` on the same window, the block registered
   on its grid, a chain collected through `on_cycle_complete`, the posterior
   against a truth you type in, and a trace plot.

## Setup

From the repository root, with h5py (either extra) and matplotlib (arrives
with `numeric-orbits`):

```sh
uv sync --extra numeric-orbits --extra examples
ENCHILADA_MOJITO_L1=/path/to/simulated_L1.h5 uv run --no-sync jupyter lab forge/block.ipynb
```

The notebooks read three environment variables:

| Variable | Meaning | Default |
| --- | --- | --- |
| `ENCHILADA_MOJITO_L1` | the Mojito-layout L1 file | required |
| `FORGE_DURATION_S` | native seconds to load | `86400` (one day) |
| `FORGE_FACTOR` | decimation factor | `10` (4 Hz to 0.4 Hz) |

`data.py` is shared by both notebooks. `load_window` goes through
`load_mojito_l1` and `decimate` and returns the window on the time,
frequency or WDM grid (for WDM, pass both division counts instead of
`duration_s`; the native duration is derived so the decimated record fits the
grid). `white_psd` and `analytic_covariance` turn a one-sided PSD, any
callable `psd(f_hz)` or one per channel, into a `Covariance` on that grid.

## What to replace and what to keep

Replace the model (`_template`, `initial_sample`, `sample`) and the checks'
expectations (the grid, what the likelihood needs from the covariance). Keep
the shape of the contract: a `BlockState` with a template on the input grid,
a complete state after a rejection too, draws from the supplied `rng`,
diagnostics on `block.state` rather than in the ledger, and a fresh instance
per campaign. Collect chains outside the block, as `wheel.ipynb` does.

To move the block to another grid, change `DOMAIN` in both notebooks, render
`_template` on that grid and update `_check_residual`.

## Caveats

- **Time grid, white noise only.** enchilada's time-grid covariance is one
  variance per sample, so `analytic_covariance` accepts only a white PSD
  there and refuses a coloured one. Coloured noise needs the frequency or
  WDM grid.
- **Edge transient.** `decimate` leaves 40 filter-transient samples at each
  end of the record (`enchilada.resample.edge_samples`); nothing is trimmed,
  and on a WDM grid the transient leaks a few columns inward.
- **PSD flat within a WDM band.** A WDM cell takes `S(f) / 2` at its band
  frequency; row 0 interleaves DC and Nyquist.
- **White stand-in.** The notebooks' `LEVEL` is a white PSD at the loaded
  window's own variance, a placeholder for the simulation's noise model, and
  a loud injected signal inflates it. The GB example's `FixedLISANoise().psd`
  (lisatools) drops in as `PSD` when you have that stack.
````
